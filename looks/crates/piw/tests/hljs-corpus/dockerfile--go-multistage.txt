# syntax=docker/dockerfile:1.7
ARG GO_VERSION=1.22
ARG ALPINE_VERSION=3.20

# ---- build stage ----
FROM golang:${GO_VERSION}-alpine${ALPINE_VERSION} AS build
RUN apk add --no-cache git ca-certificates tzdata && \
    update-ca-certificates
WORKDIR /src
COPY go.mod go.sum ./
RUN --mount=type=cache,target=/go/pkg/mod \
    go mod download  
COPY . .
ARG VERSION=dev
ARG COMMIT
RUN --mount=type=cache,target=/root/.cache/go-build \
    CGO_ENABLED=0 GOOS=linux go build \
      -trimpath \
      -ldflags "-s -w -X main.version=${VERSION} -X main.commit=${COMMIT:-unknown}" \
      -o /out/server ./cmd/server

# ---- runtime stage ----
FROM scratch AS runtime
LABEL org.opencontainers.image.title="server" \
      org.opencontainers.image.source="https://github.com/example/server" \
      org.opencontainers.image.description="Serveur HTTP — très léger 🪶" \
      maintainer="Zoë Müller <zoe@example.org>"  
COPY --from=build /etc/ssl/certs/ca-certificates.crt /etc/ssl/certs/
COPY --from=build /usr/share/zoneinfo /usr/share/zoneinfo
COPY --from=build /out/server /server
ENV TZ=Europe/Paris \
    PORT=8080
EXPOSE 8080/tcp
USER 65532:65532
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
  CMD ["/server", "-healthcheck"]
ENTRYPOINT ["/server"]
CMD ["--listen", ":8080", "--log-format=json"]
