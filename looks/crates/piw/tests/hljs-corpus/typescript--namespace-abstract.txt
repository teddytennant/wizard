/// <reference types="node" />

export namespace Shapes {
  export abstract class Shape {
    protected static count = 0;
    abstract readonly kind: string;

    constructor(public readonly label: string = 'unnamed') {
      Shape.count++;
    }

    abstract area(): number;

    describe(): string {
      return `${this.kind}(${this.label}) area=${this.area().toFixed(3)}`;
    }

    static get total(): number {
      return Shape.count;
    }
  }

  export class Circle extends Shape {
    readonly kind = 'circle';
    constructor(private r: number, label?: string) {
      super(label);
    }
    area() {
      return Math.PI * this.r ** 2;
    }
  }

  export class Rect extends Shape {
    readonly kind = 'rect';
    constructor(private w: number, private h: number, label?: string) {
      super(label);
    }
    area() {
      return this.w * this.h;
    }
  }
}

export const enum Level {
  Debug = 10,
  Info = 20,
  Warn = 30,
}

function overload(x: string): string;
function overload(x: number): number;
function overload(x: string | number): string | number {
  return typeof x === 'string' ? x.toUpperCase() : x * 2;
}
