const express = require('express');
const helmet = require('helmet');
const { randomUUID } = require('crypto');  

const app = express();
const PORT = process.env.PORT ?? 3000;
const notes = new Map();

app.use(helmet()); 
app.use(express.json({ limit: '1mb' }));

// request id + access log
app.use((req, res, next) => {
  req.id = req.get('x-request-id') || randomUUID();
  const started = process.hrtime.bigint();
  res.on('finish', () => {
    const ms = Number(process.hrtime.bigint() - started) / 1e6;
    console.log(`${req.id} ${req.method} ${req.originalUrl} -> ${res.statusCode} (${ms.toFixed(2)}ms)`);
  });
  next();
});

app.get('/api/notes', (req, res) => {
  const q = (req.query.q || '').toString().toLowerCase();
  const list = [...notes.values()].filter((n) => n.title.toLowerCase().includes(q));
  res.json({ count: list.length, items: list });
});

app.post('/api/notes', (req, res) => {
  const { title, body = '' } = req.body || {};
  if (typeof title !== 'string' || !title.trim()) {
    return res.status(400).json({ error: 'title is required' });
  }
  const note = { id: randomUUID(), title: title.trim(), body, createdAt: Date.now() };
  notes.set(note.id, note);
  res.status(201).location(`/api/notes/${note.id}`).json(note);
});

app.delete('/api/notes/:id', (req, res) => {
  if (!notes.delete(req.params.id)) return res.sendStatus(404);
  res.sendStatus(204);
});

app.use((err, req, res, _next) => {
  console.error('unhandled: %s', err.stack);
  res.status(500).json({ error: 'internal error', id: req.id });
});

app.listen(PORT, () => console.log(`listening on http://localhost:${PORT} ✓`));
