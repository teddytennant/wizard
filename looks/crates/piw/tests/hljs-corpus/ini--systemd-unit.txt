# /etc/systemd/system/myapp.service
[Unit]
Description=My App server (Café edition)
Documentation=https://example.org/docs man:myapp(8)
After=network-online.target postgresql.service
Wants=network-online.target
Requires=myapp.socket
StartLimitIntervalSec=60
StartLimitBurst=5

[Service]
Type=notify  
User=myapp
Group=myapp
DynamicUser=no
WorkingDirectory=/opt/myapp
EnvironmentFile=-/etc/default/myapp
Environment="RUST_LOG=info,myapp=debug" "LISTEN=0.0.0.0:8080"
Environment=GREETING="hello world"
ExecStartPre=/usr/bin/test -f /opt/myapp/config.toml
ExecStart=/opt/myapp/bin/myapp \
    --config /opt/myapp/config.toml \
    --workers ${WORKERS} \
    --log-format json
ExecReload=/bin/kill -HUP $MAINPID
Restart=on-failure
RestartSec=5s
TimeoutStopSec=30
LimitNOFILE=65_536
LimitNOFILE=65536
# Hardening
NoNewPrivileges=yes
ProtectSystem=strict
ProtectHome=true
PrivateTmp=true
ReadWritePaths=/var/lib/myapp /var/log/myapp
CapabilityBoundingSet=CAP_NET_BIND_SERVICE
AmbientCapabilities=CAP_NET_BIND_SERVICE
SystemCallFilter=@system-service ~@privileged   

[Install]
WantedBy=multi-user.target
Alias=app.service
