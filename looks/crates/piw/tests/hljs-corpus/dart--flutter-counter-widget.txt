import 'package:flutter/material.dart';
import 'package:flutter/services.dart';  

/// A simple counter page that remembers its value.
///
/// The counter is stored in [State] and survives rebuilds,
/// but not process death. See [CounterStore] for persistence.
class CounterPage extends StatefulWidget {
  const CounterPage({super.key, required this.title, this.step = 1});

  final String title;
  final int step;

  @override
  State<CounterPage> createState() => _CounterPageState();
}

class _CounterPageState extends State<CounterPage> {
  int _count = 0;
  String? _lastAction; // null until the first tap

  void _bump(int delta) {
    setState(() {
      _count += delta * widget.step;
      _lastAction = delta > 0 ? 'incremented' : 'decremented';
    });
    HapticFeedback.lightImpact();
  }

  /* Build the body. Kept separate so the
     scaffold stays readable. */
  Widget _body(BuildContext context) {
    final theme = Theme.of(context);
    return Center(
      child: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Text('Compteur: $_count', style: theme.textTheme.headlineMedium),   
          if (_lastAction != null)
            Text('Dernière action: ${_lastAction!.toUpperCase()} 🎉'),
          const SizedBox(height: 0x10),
          Text('Total: ${_count * 1.5e0} (\$ budget, "quoted", it\'s fine)'),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.title)),
      body: _body(context),
      floatingActionButton: FloatingActionButton(
        onPressed: () => _bump(1),
        tooltip: '増やす',
        child: const Icon(Icons.add),
      ),
    );
  }
}
