// Chained hash map keyed by C strings (FNV-1a).
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NBUCKETS 16

typedef struct entry {
    char         *key;
    int           value;
    struct entry *next;
} entry_t;

typedef struct {
    entry_t *buckets[NBUCKETS];
    size_t   count;
} map_t;

static unsigned long long fnv1a(const char *s)
{
    unsigned long long h = 0xcbf29ce484222325ULL;
    for (; *s; ++s) {
        h ^= (unsigned char)*s;
        h *= 0x100000001b3ULL;
    }
    return h;
}

void map_put(map_t *m, const char *key, int value)
{
    size_t i = fnv1a(key) % NBUCKETS;
    for (entry_t *e = m->buckets[i]; e; e = e->next) {
        if (strcmp(e->key, key) == 0) { e->value = value; return; }
    }
    entry_t *e = malloc(sizeof *e);
    e->key = strdup(key);
    e->value = value;
    e->next = m->buckets[i];
    m->buckets[i] = e;
    m->count++;
}

int map_get(const map_t *m, const char *key, int *out)
{
    for (entry_t *e = m->buckets[fnv1a(key) % NBUCKETS]; e; e = e->next)
        if (!strcmp(e->key, key)) { *out = e->value; return 1; }
    return 0;
}

int main(void)
{
    map_t m = {0};
    const char *words[] = { "naïve", "café", "日本語", "alpha", "beta", "gamma" };
    for (size_t i = 0; i < sizeof words / sizeof words[0]; i++) map_put(&m, words[i], (int)i * 11);
    int v;
    if (map_get(&m, "café", &v)) printf("café=%d (%zu keys)\n", v, m.count);
    return 0;
}
