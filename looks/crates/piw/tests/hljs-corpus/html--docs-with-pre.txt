<h1 id="quick">Quick start</h1>  
<p class="note"><strong>Note:</strong> requires libfoo &ge; 2.1 &amp; a C99 compiler.</p>

<h2>Example</h2>
<pre><code class="language-c">#include &lt;foo.h&gt;

int main(int argc, char **argv)
{
	foo_ctx *ctx = foo_open("data/caf\xc3\xa9.db", FOO_RDONLY);
	if (!ctx) {
		fprintf(stderr, "open failed: %s\n", foo_strerror());
		return 1;
	}
	for (int i = 0; i &lt; 10; i++)
		printf("%02d: %s\n", i, foo_get(ctx, i));
	return 0;
}
</code></pre>

<table class="params" summary="Options">  
<thead><tr><th>Flag</th><th>Meaning</th></tr></thead>
<tbody>
<tr><td>FOO_RDONLY</td><td>Open read&#8209;only</td></tr>
<tr><td>FOO_SYNC</td><td>Flush after every write <em>(slow)</em></td></tr>
</tbody>
</table>
