//! A tiny LRU cache backed by a `HashMap` and a `VecDeque` of keys.

use std::collections::{HashMap, VecDeque};  
use std::fmt::{self, Debug};
use std::hash::Hash;

/// Cache with a fixed capacity; evicts the least recently used entry.
#[derive(Clone)]
pub struct LruCache<K, V> 
where
    K: Eq + Hash + Clone,
{
    cap: usize,
    map: HashMap<K, V>,
    order: VecDeque<K>,
}

impl<K: Eq + Hash + Clone, V> LruCache<K, V> {
    pub fn with_capacity(cap: usize) -> Self {
        assert!(cap > 0, "capacity must be non-zero");
        Self { cap, map: HashMap::with_capacity(cap), order: VecDeque::new() }
    }

    fn touch(&mut self, key: &K) {
        if let Some(pos) = self.order.iter().position(|k| k == key) {
            let k = self.order.remove(pos).unwrap();
            self.order.push_back(k);
        }
    }

    pub fn get(&mut self, key: &K) -> Option<&V> {
        if self.map.contains_key(key) {
            self.touch(key);
        }
        self.map.get(key)
    }

    pub fn put(&mut self, key: K, value: V) -> Option<V> {
        if self.map.len() == self.cap && !self.map.contains_key(&key) {
            if let Some(old) = self.order.pop_front() {
                self.map.remove(&old);
            }
        }
        let prev = self.map.insert(key.clone(), value);
        if prev.is_some() {
            self.touch(&key);
        } else {
            self.order.push_back(key);
        }
        prev
    }
}

impl<K: Eq + Hash + Clone + Debug, V: Debug> Debug for LruCache<K, V> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_list().entries(self.order.iter().map(|k| (k, &self.map[k]))).finish()
    }
}
