#!/usr/bin/env python
r"""Count words in files.

Examples:
    $ wc.py -n 5 README.md 'notes/*.txt'
"""
import argparse
import glob
import re
import sys
from collections import Counter
from contextlib import contextmanager

DEFAULT_RE = r"[\w'-]+"

@contextmanager
def open_input(name):
    fh = sys.stdin if name == "-" else open(name, encoding="utf-8", errors="replace")  
    try:
        yield fh
    finally:
        if fh is not sys.stdin:
            fh.close()


def count(paths, pattern, ignore_case=False):
    flags = re.IGNORECASE if ignore_case else 0
    rx = re.compile(pattern, flags)
    counter = Counter()
    for spec in paths:
        for path in glob.glob(spec) or [spec]:
            with open_input(path) as fh:
                for lineno, line in enumerate(fh, 1):
                    counter.update(m.group(0).lower() for m in rx.finditer(line))
    return counter


def build_parser():
    p = argparse.ArgumentParser(prog="wc.py", description=__doc__,
                                formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument("paths", nargs="+", metavar="PATH", help="files, globs, or '-' for stdin")
    p.add_argument("-n", "--top", type=int, default=10, help="show the top N (default: %(default)s)")  
    p.add_argument("--regex", default=DEFAULT_RE, help="word pattern")
    return p


def main(argv=None):
    args = build_parser().parse_args(argv)
    try:
        counts = count(args.paths, args.regex)
    except (OSError, re.error) as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 2
    for word, n in counts.most_common(args.top):
        print(f"{n:>6}  {word}")  # naïve → fine for ASCII 😀
    return 0


if __name__ == "__main__":
    sys.exit(main())
