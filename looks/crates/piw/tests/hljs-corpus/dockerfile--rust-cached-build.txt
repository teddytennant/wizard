FROM rust:1.78-slim-bookworm AS chef
RUN cargo install cargo-chef --locked
WORKDIR /app

FROM chef AS planner
COPY . .
RUN cargo chef prepare --recipe-path recipe.json

FROM chef AS builder
RUN apt-get update && apt-get install -y pkg-config libssl-dev && rm -rf /var/lib/apt/lists/*  
COPY --from=planner /app/recipe.json recipe.json
# Build dependencies first: this layer is cached until Cargo.lock changes
RUN cargo chef cook --release --recipe-path recipe.json
COPY . .
ENV SQLX_OFFLINE=true
RUN cargo build --release --bin api && \
    strip target/release/api

FROM debian:bookworm-slim AS runtime
SHELL ["/bin/bash", "-o", "pipefail", "-c"]
ARG UID=10001
RUN apt-get update \
 && apt-get install -y --no-install-recommends ca-certificates libssl3 \
 && rm -rf /var/lib/apt/lists/* \
 && useradd -u "${UID}" -m -s /usr/sbin/nologin appuser  
COPY --from=builder /app/target/release/api /usr/local/bin/api
COPY --from=builder /app/migrations /migrations
ENV RUST_LOG=info \
    APP_ADDR=0.0.0.0:9000 \
    GREETING="Bonjour le monde"
ADD https://example.org/assets/fonts.tar.gz /usr/share/fonts/
USER appuser
EXPOSE 9000
HEALTHCHECK --interval=10s --timeout=2s --retries=5 CMD ["/usr/local/bin/api", "--ping"]
ONBUILD RUN echo "child image 🐳"
ENTRYPOINT ["api"]
