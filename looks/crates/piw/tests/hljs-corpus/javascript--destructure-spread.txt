  theme: 'dark',

  telemetry: null,
};

function mergeConfig(base, ...overrides) {
  return overrides.reduce(
    (acc, o) => ({
      ...acc,
      ...o,
      editor: { ...acc.editor, ...(o?.editor ?? {}) },
    }),
    base,
  );
}
