#!/usr/bin/env bash
# Generate a systemd unit and an nginx vhost from a few variables.

set -eo pipefail

APP_NAME=${1:?usage: $0 APP_NAME [PORT]}
PORT=${2:-8080}
APP_USER="svc-${APP_NAME//[^a-z0-9]/}"
UNIT="/etc/systemd/system/${APP_NAME}.service"
HOST="${APP_NAME}.example.com"

if ! id -u "$APP_USER" &>/dev/null; then
	useradd --system --no-create-home --shell /usr/sbin/nologin "$APP_USER"
fi

tee "$UNIT" >/dev/null <<UNIT_EOF
[Unit]
Description=${APP_NAME} (managed, generated $(date -u +%FT%TZ))
After=network-online.target

[Service]
User=${APP_USER}
Environment="PORT=${PORT}" "NODE_ENV=production"
ExecStart=/opt/${APP_NAME}/bin/server --listen 127.0.0.1:\${PORT}
Restart=on-failure
RestartSec=$((PORT % 7 + 3))

[Install]
WantedBy=multi-user.target
UNIT_EOF

# quoted delimiter: nothing below is expanded by the shell
cat > "/etc/nginx/conf.d/${APP_NAME}.conf" <<'NGINX'
server {
    listen 443 ssl http2;
    server_name $host;   # filled in by envsubst
    location / {
        proxy_pass http://127.0.0.1:$port;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
NGINX
host=$HOST port=$PORT envsubst '$host $port' < "/etc/nginx/conf.d/${APP_NAME}.conf" | sponge "/etc/nginx/conf.d/${APP_NAME}.conf"

cat <<-'MSG'
	Installed. Next steps:
	  systemctl daemon-reload
	  systemctl enable --now "$APP_NAME"
MSG

nginx -t 2>&1 | grep -qi 'successful' && systemctl reload nginx || echo "nginx config invalid!" >&2
