import { EventEmitter } from 'node:events';
import { setTimeout as sleep } from 'node:timers/promises';

const DEFAULTS = { concurrency: 4, retries: 3, backoffMs: 250 };

export class TaskQueue extends EventEmitter {
	static #instances = 0;
	#pending = [];
	#running = 0;
	#opts;

	constructor(opts = {}) {
		super();
		this.#opts = { ...DEFAULTS, ...opts };
		TaskQueue.#instances++;
	}

	static get count() {
		return TaskQueue.#instances;
	}

	push(fn, label = fn.name || '<anonymous>') {
		return new Promise((resolve, reject) => {
			this.#pending.push({ fn, label, resolve, reject, attempt: 0 });
			queueMicrotask(() => this.#drain());
		});
	}

	async #run(job) {
		this.#running++;
		try {
			while (true) {
				try {
					const value = await job.fn();
					this.emit('done', job.label, value);
					return job.resolve(value);
				} catch (err) {
					if (++job.attempt > this.#opts.retries) throw err;
					const delay = this.#opts.backoffMs * 2 ** (job.attempt - 1);
					this.emit('retry', job.label, job.attempt, delay);
					await sleep(delay);
				}
			}
		} catch (err) {
			job.reject(err);
		} finally {
			this.#running--;
			this.#drain();
		}
	}

	#drain() {
		while (this.#running < this.#opts.concurrency && this.#pending.length) {
			void this.#run(this.#pending.shift());
		}
		if (!this.#running && !this.#pending.length) this.emit('idle');
	}
}
