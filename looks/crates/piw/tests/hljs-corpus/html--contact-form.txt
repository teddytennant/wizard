<script>
  document.querySelector('form.contact').addEventListener('submit', function (ev) {
    var email = this.elements['email'].value;
    if (!/^[^@\s]+@[^@\s]+\.\w{2,}$/.test(email)) {
      ev.preventDefault();
      var err = document.getElementById('err');
      err.textContent = 'Adresse invalide: "' + email + '"';
      err.hidden = false;
    }
  });
