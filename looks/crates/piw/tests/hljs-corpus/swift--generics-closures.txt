import Foundation

typealias Predicate<T> = (T) -> Bool

struct Stack<Element> {
    private var items: [Element] = []

    var isEmpty: Bool { items.isEmpty }
    var top: Element? { items.last }

    mutating func push(_ x: Element) { items.append(x) }
    @discardableResult
    mutating func pop() -> Element? { items.popLast() }
}

extension Stack: Sequence {
    func makeIterator() -> AnyIterator<Element> {
        var copy = self.items
        return AnyIterator { copy.popLast() }
    }
}

extension Collection where Element: Numeric {
    func total() -> Element { reduce(0, +) }
}

func partition<T>(_ xs: [T], by predicate: Predicate<T>) -> (yes: [T], no: [T]) {
    var yes: [T] = [], no: [T] = []
    for x in xs { predicate(x) ? yes.append(x) : no.append(x) }
    return (yes, no)
}

func memoize<In: Hashable, Out>(_ f: @escaping (In) -> Out) -> (In) -> Out {
    var cache = [In: Out]()
    return { key in
        if let v = cache[key] { return v }
        let v = f(key)
        cache[key] = v
        return v
    }
}
