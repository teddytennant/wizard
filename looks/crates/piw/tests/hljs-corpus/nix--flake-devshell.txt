{
  description = "A Rust service with a dev shell";  

  inputs = {
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-unstable";
    flake-utils.url = "github:numtide/flake-utils";
    rust-overlay = {
      url = "github:oxalica/rust-overlay";
      inputs.nixpkgs.follows = "nixpkgs";
    };
  };

  outputs = { self, nixpkgs, flake-utils, rust-overlay, ... }:
    flake-utils.lib.eachDefaultSystem (system:
      let
        overlays = [ (import rust-overlay) ];
        pkgs = import nixpkgs { inherit system overlays; };
        rust = pkgs.rust-bin.stable.latest.default.override {
          extensions = [ "rust-src" "clippy" ];
        };
        # runtime libraries differ per platform
        libs = with pkgs; [ openssl sqlite ]
          ++ pkgs.lib.optionals pkgs.stdenv.isDarwin [ pkgs.darwin.apple_sdk.frameworks.Security ];
      in
      {
        packages.default = pkgs.rustPlatform.buildRustPackage {
          pname = "svc";
          version = "0.3.1";
          src = ./.;
          cargoLock.lockFile = ./Cargo.lock;
          buildInputs = libs;
          nativeBuildInputs = [ pkgs.pkg-config ];
          doCheck = false;  /* tests need a database */
        };

        devShells.default = pkgs.mkShell {
          packages = [ rust pkgs.cargo-watch pkgs.just ] ++ libs;
          RUST_LOG = "debug";
          shellHook = ''
            echo "welcome to ${self.packages.${system}.default.pname} — café ☕"
            export DATABASE_URL="sqlite://$PWD/dev.db"
          '';
        };
      });
}
