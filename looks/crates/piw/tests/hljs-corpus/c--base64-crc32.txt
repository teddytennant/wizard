/* codec.c - base64 and CRC-32 for the upload path */
#include <stdint.h>
#include <stdio.h>
#include <string.h>

static const char B64[] =
	"ABCDEFGHIJKLMNOPQRSTUVWXYZ"
	"abcdefghijklmnopqrstuvwxyz"
	"0123456789+/";

static uint32_t crc_table[256];

static void crc_init(void)
{
	for (uint32_t i = 0; i < 256; i++) {
		uint32_t c = i;
		for (int k = 0; k < 8; k++)
			c = (c & 1) ? 0xEDB88320u ^ (c >> 1) : c >> 1;
		crc_table[i] = c;
	}
}

uint32_t crc32(const uint8_t *p, size_t n)
{
	uint32_t c = ~0u;
	while (n--) c = crc_table[(c ^ *p++) & 0xFF] ^ (c >> 8);
	return ~c;
}

size_t b64_encode(const uint8_t *in, size_t n, char *out)
{
	char *o = out;
	for (; n >= 3; n -= 3, in += 3) {
		*o++ = B64[in[0] >> 2];
		*o++ = B64[((in[0] & 0x03) << 4) | (in[1] >> 4)];
		*o++ = B64[((in[1] & 0x0F) << 2) | (in[2] >> 6)];
		*o++ = B64[in[2] & 0x3F];
	}
	if (n) {
		*o++ = B64[in[0] >> 2];
		*o++ = B64[((in[0] & 0x03) << 4) | (n > 1 ? in[1] >> 4 : 0)];
		*o++ = n > 1 ? B64[(in[1] & 0x0F) << 2] : '=';
		*o++ = '=';
	}
	*o = '\0';
	return (size_t)(o - out);
}

int main(void)
{
	const char *msg = "héllo wörld \xE2\x9C\x93";   // UTF-8 bytes
	char enc[64];
	crc_init();
	size_t len = b64_encode((const uint8_t *)msg, strlen(msg), enc);
	printf("%s (%zu chars) crc=%08x\n", enc, len, crc32((const uint8_t *)msg, strlen(msg)));
	return 0;
}
